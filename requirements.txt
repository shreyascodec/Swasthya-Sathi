# Phase 1 — Intake & Preprocess (classical CV; no GPU, ARM-compatible)
#
# UPPER BOUNDS ARE LOAD-BEARING, not caution. opencv-python-headless 5.x requires
# numpy>=2, while the verified stack (paddlepaddle 3.x + torch 2.6+cu126) runs on
# numpy 1.26.4. With open ranges, `pip install -r requirements.txt` on a fresh
# kiosk resolves opencv 5 against numpy 1.26 and `pip check` fails — the box that
# was tested and the box that ships stop being the same machine.
opencv-python-headless>=4.9,<5   # headless = no GUI libs; works on kiosk/cloud/ARM
numpy>=1.26,<2

# Phase 2 — OCR (real engine). PaddleOCR 3.x + paddlepaddle.
#
# GPU STRONGLY RECOMMENDED (ocrplan.md / report.md §4.4a): the CPU wheel below
# runs OCR ~60 s per 3-page session vs <1 s of GPU inference. Install the GPU
# build instead, matching your CUDA:
#     pip uninstall paddlepaddle
#     pip install paddlepaddle-gpu==3.3.1 \
#       -i https://www.paddlepaddle.org.cn/packages/stable/cu126/
#
# WINDOWS CUDA/cuDNN CONFLICT: paddle and torch each bundle their own cuDNN
# DLLs under the SAME filenames. If their cuDNN majors/minors differ, whichever
# imports second dies with "WinError 127 ... cudnn_cnn64_9.dll". Keep them on
# the same CUDA line — this repo runs paddlepaddle-gpu cu126 + torch cu126
# (both cuDNN 9.5.x); install torch with:
#     pip install torch==2.6.0+cu126 torchvision==0.21.0+cu126 \
#       --index-url https://download.pytorch.org/whl/cu126
#
# On CPU-only hosts the adapter keeps enable_mkldnn=False to avoid a
# paddlepaddle 3.x oneDNN crash on Windows (flip ocr.primary.enable_mkldnn in
# config/models.yaml where oneDNN works — worth ~2-3x on CPU).
paddleocr>=3.7
paddlepaddle>=3.3
